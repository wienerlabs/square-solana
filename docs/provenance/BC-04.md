sızdırıyordu):

```bash
git clone --no-local ~/square-planning/sources/covenant <geçici-dizin>   # tam geçmiş, shallow değil
cd <geçici-dizin>
git filter-repo --path programs/covenant/ --path tests/covenant.ts \
  --path-rename programs/covenant/:programs/square-escrow/ \
  --path-rename tests/covenant.ts:tests/legacy/covenant.ts
# square-solana'da:
git fetch <geçici-dizin> HEAD
git merge --allow-unrelated-histories --no-ff FETCH_HEAD
```
